name: reject-port-unreachable
about: REJECT with no option sends ICMP port unreachable. The sender hears "refused" at once,
about: for TCP and UDP alike, even though a program is listening behind the rule.
--- iptables
*filter
:INPUT ACCEPT [0:0]
:FORWARD ACCEPT [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m tcp --dport 80 -j REJECT
-A INPUT -p udp -m udp --dport 53 -j REJECT --reject-with icmp-port-unreachable
COMMIT
--- listen
host tcp 0.0.0.0 80
host udp 0.0.0.0 53
host tcp 0.0.0.0 22
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 80 want:refused
eth0 198.51.100.7 203.0.113.10 udp 53 want:refused
eth0 198.51.100.7 203.0.113.10 tcp 22 want:open
