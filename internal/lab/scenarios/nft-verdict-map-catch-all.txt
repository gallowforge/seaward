name: nft-verdict-map-catch-all
about: A verdict map in the middle of a rule, with a catch-all entry. A port the map names with
about: "continue" goes on to the source test that follows it; every other port takes the
about: catch-all's verdict, so the rest of the rule never runs for it. A second map has no
about: catch-all: a port it does not hold ends that rule and falls to the next one.
--- nft
table inet fw {
	chain input {
		type filter hook input priority 0; policy drop;
		ct state established,related accept
		tcp dport vmap { 22 : continue, 80 : accept, * : jump rest } ip saddr 198.51.100.0/24 accept
		tcp dport 22 reject with tcp reset
	}
	chain rest {
		tcp dport vmap { 443 : continue, 8443 : drop } ip saddr 203.0.113.0/24 accept
		tcp dport 8080 accept
	}
}
--- listen
host tcp 0.0.0.0 22
host tcp 0.0.0.0 80
host tcp 0.0.0.0 443
host tcp 0.0.0.0 8080
host tcp 0.0.0.0 8443
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 22 want:open
eth0 203.0.113.77 203.0.113.10 tcp 22 want:refused
eth0 203.0.113.77 203.0.113.10 tcp 80 want:open
eth0 203.0.113.77 203.0.113.10 tcp 443 want:open
eth0 198.51.100.7 203.0.113.10 tcp 443 want:silent
eth0 203.0.113.77 203.0.113.10 tcp 8443 want:silent
eth0 198.51.100.7 203.0.113.10 tcp 8080 want:open
eth0 198.51.100.7 203.0.113.10 tcp 9999 want:silent
