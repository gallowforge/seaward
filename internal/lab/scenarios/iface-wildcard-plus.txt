name: iface-wildcard-plus
about: A trailing + makes -i a prefix match: eth+ is every interface starting with eth,
about: eth1+ is eth1 and anything longer, but not eth0.
--- iptables
*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -i eth1+ -p tcp -m tcp --dport 22 -j ACCEPT
-A INPUT -i eth+ -p tcp -m tcp --dport 80 -j ACCEPT
-A INPUT -i et -p tcp -m tcp --dport 443 -j ACCEPT
COMMIT
--- listen
host tcp 0.0.0.0 22
host tcp 0.0.0.0 80
host tcp 0.0.0.0 443
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 22 want:silent
eth1 192.0.2.50 10.20.0.1 tcp 22 want:open
eth0 198.51.100.7 203.0.113.10 tcp 80 want:open
eth1 192.0.2.50 10.20.0.1 tcp 80 want:open
eth0 198.51.100.7 203.0.113.10 tcp 443 want:silent
