name: address-on-loopback
about: A spare address held on the loopback interface, as a machine with a floating or anycast
about: address has, is the machine's own: the kernel takes a packet for it on whichever interface
about: it arrives, and the rules judge it by that address. A service bound to it answers there and
about: nowhere else, and one bound to every address answers there too.
loopback: 203.0.113.99
loopback: 2001:db8:1::99
--- iptables
*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m tcp --dport 80 -j ACCEPT
-A INPUT -d 203.0.113.99/32 -p tcp -m tcp --dport 81 -j REJECT --reject-with tcp-reset
-A INPUT -d 203.0.113.99/32 -p tcp -m tcp --dport 22 -j ACCEPT
COMMIT
--- listen
host tcp 203.0.113.99 80
host tcp 0.0.0.0 22
host tcp 0.0.0.0 81
host tcp 2001:db8:1::99 80 v6only
--- probes
eth0 198.51.100.7 203.0.113.99 tcp 80 want:open
eth0 198.51.100.7 203.0.113.10 tcp 80 want:refused
eth0 203.0.113.50 203.0.113.99 tcp 80 want:open
eth1 192.0.2.50 203.0.113.99 tcp 80 want:open
eth0 198.51.100.7 203.0.113.99 tcp 22 want:open
eth0 198.51.100.7 203.0.113.10 tcp 22 want:silent
eth0 198.51.100.7 203.0.113.99 tcp 81 want:refused
eth0 198.51.100.7 203.0.113.10 tcp 81 want:silent
eth0 2001:db8:99::7 2001:db8:1::99 tcp 80 want:open
eth0 2001:db8:99::7 2001:db8:1::10 tcp 80 want:refused
