name: nft-dnat-behind-jump
about: The same in nftables: a jump, a goto and a verdict map each lead from the rule that names
about: the port to a chain that rewrites the destination, one of them through a second chain. An
about: entry of the map that says accept rewrites nothing, and neither does a port the map has no
about: entry for. A port named again in the chain jumped to has to pass both rules.
--- nft
table ip nat {
	chain onward {
		tcp dport 8005 return
		meta l4proto tcp redirect to :9000
	}
	chain further {
		jump inner
	}
	chain inner {
		meta l4proto tcp dnat to 203.0.113.10:9001
	}
	chain narrow {
		tcp dport 8305 redirect to :9000
		tcp dport 8400 redirect to :9000
	}
	chain prerouting {
		type nat hook prerouting priority dstnat; policy accept;
		tcp dport 8000-8010 jump onward
		tcp dport 8100 goto further
		tcp dport vmap { 8200 : jump onward, 8201 : goto further, 8202 : accept }
		tcp dport 8300-8310 jump narrow
	}
}
table inet filter {
	chain input {
		type filter hook input priority filter; policy drop;
		tcp dport { 9000, 9001 } accept
		tcp dport { 8005, 8202, 8304, 8400 } reject with tcp reset
	}
}
--- listen
host tcp 0.0.0.0 9000
host tcp 0.0.0.0 9001
