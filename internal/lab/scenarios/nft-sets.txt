name: nft-sets
about: Named sets with intervals, a concatenated protocol . port set, an anonymous set with a
about: range, and a negated set lookup.
--- nft
table inet filter {
	set allowed4 {
		type ipv4_addr
		flags interval
		elements = { 198.51.100.0/25, 192.0.2.50 }
	}
	set allowed6 {
		type ipv6_addr
		elements = { 2001:db8:99::7 }
	}
	set svc {
		type inet_proto . inet_service
		elements = { tcp . 443, udp . 53 }
	}
	chain input {
		type filter hook input priority filter; policy drop;
		ip saddr @allowed4 tcp dport 22 accept
		ip6 saddr @allowed6 tcp dport 22 accept
		meta l4proto . th dport @svc accept
		tcp dport { 80, 8000-8100 } accept
		ip saddr != @allowed4 tcp dport 9000 accept
	}
}
--- listen
host tcp :: 22 dual
host tcp 0.0.0.0 80
host tcp 0.0.0.0 443
host tcp 0.0.0.0 8050
host tcp 0.0.0.0 9000
host udp 0.0.0.0 53
