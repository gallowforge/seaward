name: redirect-to-other-port
about: REDIRECT rewrites the destination to the arriving interface's own address and the given
about: port, before filter runs: filter sees port 22, not 8080. A packet on eth1 is redirected to
about: eth1's address, so a socket bound only to eth0's address does not get it.
--- iptables
*nat
:PREROUTING ACCEPT [0:0]
:INPUT ACCEPT [0:0]
:OUTPUT ACCEPT [0:0]
:POSTROUTING ACCEPT [0:0]
-A PREROUTING -p tcp -m tcp --dport 8080 -j REDIRECT --to-ports 22
-A PREROUTING -p tcp -m tcp --dport 2200 -j REDIRECT --to-ports 2222
COMMIT
*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m tcp --dport 22 -j ACCEPT
-A INPUT -p tcp -m tcp --dport 2222 -j ACCEPT
-A INPUT -p tcp -m tcp --dport 8080 -j REJECT
COMMIT
--- listen
host tcp 0.0.0.0 22
host tcp 0.0.0.0 8080
host tcp 203.0.113.10 2222
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 8080 want:open
eth0 198.51.100.7 203.0.113.10 tcp 22 want:open
eth0 198.51.100.7 203.0.113.10 tcp 8081 want:silent
eth0 198.51.100.7 203.0.113.10 tcp 2200 want:open
eth1 192.0.2.50 10.20.0.1 tcp 2200 want:refused
eth0 198.51.100.7 10.20.0.1 tcp 2200 want:open
