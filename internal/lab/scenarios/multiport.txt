name: multiport
about: -m multiport --dports takes a list with ranges inside it; ! negates the whole list.
--- iptables
*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m multiport --dports 22,80,8000:8100 -j ACCEPT
-A INPUT -p udp -m multiport ! --dports 53,123 -j ACCEPT
COMMIT
