name: rule-without-target
about: A rule with no target only counts packets; the walk goes on to the next rule.
--- iptables
*filter
:INPUT ACCEPT [0:0]
:FORWARD ACCEPT [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m tcp --dport 22
-A INPUT -p tcp -m tcp --dport 22 -j DROP
-A INPUT -p tcp -m tcp --dport 80
COMMIT
--- listen
host tcp 0.0.0.0 22
host tcp 0.0.0.0 80
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 22 want:silent
eth0 198.51.100.7 203.0.113.10 tcp 80 want:open
