name: nft-netdev-ingress
about: A chain on the ingress hook of a netdev table runs for each packet as it arrives on its
about: interface, before every other chain. A drop there is final whatever the input chain allows,
about: a port match holds for IPv4 and IPv6 alike, and the chain sees nothing that arrives on
about: another interface.
--- nft
table netdev early {
	set banned {
		type ipv4_addr
		elements = { 198.51.100.9 }
	}
	chain ingress {
		type filter hook ingress device "eth0" priority -500; policy accept;
		tcp dport 80 drop
		ip saddr @banned drop
		ip6 saddr 2001:db8:99::9 udp dport 53 drop
	}
}
table inet filter {
	chain input {
		type filter hook input priority filter; policy drop;
		tcp dport { 22, 80 } accept
		udp dport 53 accept
	}
}
--- listen
host tcp :: 22 dual
host tcp :: 80 dual
host udp :: 53 dual
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 22 want:open
eth0 198.51.100.7 203.0.113.10 tcp 80 want:silent
eth0 2001:db8:99::7 2001:db8:1::10 tcp 22 want:open
eth0 2001:db8:99::7 2001:db8:1::10 tcp 80 want:silent
eth0 198.51.100.9 203.0.113.10 tcp 22 want:silent
eth1 192.0.2.50 10.20.0.1 tcp 80 want:open
eth0 2001:db8:99::9 2001:db8:1::10 udp 53 want:silent
eth0 2001:db8:99::7 2001:db8:1::10 udp 53 want:open
