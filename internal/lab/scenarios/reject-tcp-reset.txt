name: reject-tcp-reset
about: REJECT --reject-with tcp-reset answers a TCP connection with a reset, exactly like a
about: closed port, though a program is listening.
--- iptables
*filter
:INPUT ACCEPT [0:0]
:FORWARD ACCEPT [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -p tcp -m tcp --dport 80 -j REJECT --reject-with tcp-reset
COMMIT
--- listen
host tcp 0.0.0.0 80
host tcp 0.0.0.0 443
--- probes
eth0 198.51.100.7 203.0.113.10 tcp 80 want:refused
eth0 198.51.100.7 203.0.113.10 tcp 443 want:open
eth1 192.0.2.50 10.20.0.1 tcp 80 want:refused
